        <h3>Zelf een game uploaden?</h3>

    <script>
        // Jouw persoonlijke Supabase API-sleutels
        const SUPABASE_URL = "https://supabase.co";
        const SUPABASE_KEY = "sb_publishable_Hrkm_ghr_xNpXdCYITo1Ug_F4assrx8";
        
        // Verbinden met de database
        const supabase = supabasejs.createClient(SUPABASE_URL, SUPABASE_KEY);

        // 1. Functie om alle games op te halen en te tonen
        async function laadGames() {
            const { data, error } = await supabase.storage.from('games').list();
            if (error) {
                console.error("Fout bij laden van games:", error.message);
                return;
            }
            
            const lijst = document.getElementById('gameLijst');
            lijst.innerHTML = ""; 

            data.forEach(file => {
                if(file.name === ".emptyFolderPlaceholder") return;
                
                const { data: urlData } = supabase.storage.from('games').getPublicUrl(file.name);
                
                // Haal de tags uit de bestandsnaam (als die erin zitten)
                let schoneNaam = file.name.replace('.html', '');
                let isPC = schoneNaam.includes('_PC');
                let isMobiel = schoneNaam.includes('_MOB');
                
                // Maak de naam weer netjes voor het scherm
                schoneNaam = schoneNaam.replace('_PC', '').replace('_MOB', '');

                // BELANGRIJK: sandbox="allow-scripts allow-same-origin" zorgt ervoor dat savegames bewaard blijven!
                lijst.innerHTML += `
                    <div class="card">
                        <div class="game-wrapper">
                            <iframe src="${urlData.publicUrl}" sandbox="allow-scripts allow-same-origin"></iframe>
                        </div>
                        <h3>${schoneNaam}</h3>
                        <div>
                            ${isPC ? '<span class="tag">🖥 PC</span>' : ''}
                            ${isMobiel ? '<span class="tag">📱 Mobiel</span>' : ''}
                        </div>
                    </div>
                `;
            });
        }

        // 2. Uploadfunctie die ook de PC/Telefoon tags meestuurt
        async function uploadDeGame() {
            const naamInput = document.getElementById('gameNaam').value.trim();
            const fileInput = document.getElementById('gameBestand');
            const checkPC = document.getElementById('checkPC').checked;
            const checkMobiel = document.getElementById('checkMobiel').checked;

            if (!naamInput || !fileInput.files.length) {
                alert("Vul een naam in en kies een .html game-bestand!");
                return;
            }

            const bestand = fileInput.files[0];
            
            // Plak stiekem tags aan de bestandsnaam om te onthouden waar de game geschikt voor is
            let bestandsNaam = naamInput;
            if (checkPC) bestandsNaam += "_PC";
            if (checkMobiel) bestandsNaam += "_MOB";
            bestandsNaam += ".html";

            alert("Bezig met uploaden... even geduld.");

            const { data, error } = await supabase.storage
                .from('games')
                .upload(bestandsNaam, bestand, { 
                    contentType: 'text/html',
                    upsert: true 
                });

            if (error) {
                alert("Er ging iets mis met de upload: " + error.message);
            } else {
                alert("Je game staat live op de website!");
                // Maak het formulier weer leeg
                document.getElementById('gameNaam').value = "";
                fileInput.value = "";
                document.getElementById('checkPC').checked = false;
                document.getElementById('checkMobiel').checked = false;
                laadGames(); 
            }
        }

        // Start direct als de pagina opent
        laadGames();
    </script>
